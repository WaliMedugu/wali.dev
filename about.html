Code Quality Sample

This snippet showcases how I write clean, maintainable architecture in Dart. No sloppy global state, clear separations of concerns, and robust client-side routing logic. 

abstract class DebatesDataSource {
  Future<List<DebateModel>> getTrendingDebates({required int limit});
  Future<void> voteOnDebate({required String debateId, required bool isFor});
}

class SupabaseDebatesDataSourceImpl implements DebatesDataSource {
  final SupabaseClient _client;
  SupabaseDebatesDataSourceImpl(this._client);

  @override
  Future<List<DebateModel>> getTrendingDebates({required int limit}) async {
    final response = await _client
        .from('debates')
        .select('*, profiles(*), votes(*)')
        .order('trending_score', descending: true)
        .limit(limit);
    return response.map((data) => DebateModel.fromJson(data)).toList();
  }
}
